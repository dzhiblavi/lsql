    add_hardening_test(union_parallel query.sql ${PROFILE} ${SEED}
        --order unordered --consumer full)
    add_hardening_test(merge_sorted_parallel query.sql ${PROFILE} ${SEED}
        --order key-desc --consumer full)
    add_hardening_test(union_parallel_limit early.sql ${PROFILE} ${SEED}
        --order unordered --consumer limit_17=17)
    add_hardening_test(union_parallel_limit query.sql ${PROFILE} ${SEED}
        --order unordered --consumer limit_1=1 --consumer limit_17=17 --consumer limit_1000=1000 --consumer full)
    add_hardening_test(merge_sorted_parallel_limit early.sql ${PROFILE} ${SEED}
        --order key-id-desc --consumer limit_1=1)
    add_hardening_test(merge_sorted_parallel_limit query.sql ${PROFILE} ${SEED}
        --order key-id-asc --consumer limit_17=17 --consumer limit_1000=1000 --consumer full)
endforeach()
